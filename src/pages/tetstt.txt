<script setup>
import { ref, computed, watch } from 'vue';
import { useQuasar } from 'quasar';
import PDFExport from '../components/PrintReport.vue'; // Adjust path as needed

const $q = useQuasar();

// Form data
const reportTitle = ref('');
const reportSubtitle = ref('');
const tableTitle = ref('');
const dataInputMethod = ref('json');
const jsonData = ref('');
const columnsDefinition = ref('');
const rowsData = ref('');
const includeNotes = ref(false);
const notesTitle = ref('');
const notesData = ref('');

// Validation states
const isJsonValid = ref(null);
const isKeyValueValid = ref(null);
const jsonValidationError = ref('');
const keyValueValidationError = ref('');

// Final processed data
const finalData = ref([]);
const finalColumns = ref([]);

// Computed property to check if form is valid
const isFormValid = computed(() => {
  const basicInfoValid = !!reportTitle.value && !!tableTitle.value;
  const dataValid = (dataInputMethod.value === 'json' && isJsonValid.value === true) ||
                    (dataInputMethod.value === 'keyValue' && isKeyValueValid.value === true);
  const notesValid = !includeNotes.value || (includeNotes.value && !!notesTitle.value && !!notesData.value);

  return basicInfoValid && dataValid && notesValid;
});

// Watch for changes in input method to reset validation
watch(dataInputMethod, () => {
  if (dataInputMethod.value === 'json') {
    isKeyValueValid.value = null;
  } else {
    isJsonValid.value = null;
  }
});

// JSON validation function
const validateJson = (jsonString) => {
  try {
    if (!jsonString) return false;
    const parsed = JSON.parse(jsonString);
    return true;
  } catch (error) {
    return false;
  }
};

// Check for duplicate IDs in data
const hasDuplicateIds = (data) => {
  if (!Array.isArray(data) || data.length <= 1) return false;

  const ids = new Set();

  for (const item of data) {
    if (item.id !== undefined) {
      if (ids.has(item.id)) {
        return true; // Found duplicate ID
      }
      ids.add(item.id);
    }
  }

  return false;
};

// Validate consistency in JSON data
const validateJsonConsistency = (parsedData) => {
  // Convert single object to array for consistent processing
  const data = Array.isArray(parsedData) ? parsedData : [parsedData];

  if (data.length === 0) {
    return { valid: false, error: 'Data cannot be empty' };
  }

  // Check for duplicate IDs
  if (hasDuplicateIds(data)) {
    return { valid: false, error: 'Data contains duplicate ID values' };
  }

  // For a single object, no further consistency check needed
  if (data.length === 1) {
    return { valid: true };
  }

  // Get keys from first object as reference
  const firstObjectKeys = Object.keys(data[0]);

  // Check that all objects have the same keys
  for (let i = 1; i < data.length; i++) {
    const currentKeys = Object.keys(data[i]);

    // Check if current object has all required keys
    const missingKeys = firstObjectKeys.filter(key => !currentKeys.includes(key));
    if (missingKeys.length > 0) {
      return {
        valid: false,
        error: `Row ${i+1} is missing required fields: ${missingKeys.join(', ')}`
      };
    }

    // Check for extra keys not in the first object
    const extraKeys = currentKeys.filter(key => !firstObjectKeys.includes(key));
    if (extraKeys.length > 0) {
      return {
        valid: false,
        error: `Row ${i+1} has extra fields not present in all rows: ${extraKeys.join(', ')}`
      };
    }
  }

  return { valid: true };
};

// Validate and process JSON data
const validateAndPreviewJson = () => {
  try {
    const parsed = JSON.parse(jsonData.value);
    const dataArray = Array.isArray(parsed) ? parsed : [parsed];
    const consistencyCheck = validateJsonConsistency(parsed);

    if (consistencyCheck.valid) {
      isJsonValid.value = true;
      jsonValidationError.value = '';

      // Create columns automatically from the first row
      if (dataArray.length > 0) {
        const firstRow = dataArray[0];
        finalColumns.value = Object.keys(firstRow).map(key => ({
          name: key,
          label: key.charAt(0).toUpperCase() + key.slice(1).replace(/([A-Z])/g, ' $1'),
          field: key,
          sortable: true
        }));

        finalData.value = dataArray;
      }
    } else {
      isJsonValid.value = false;
      jsonValidationError.value = consistencyCheck.error;
    }
  } catch (error) {
    isJsonValid.value = false;
    jsonValidationError.value = 'Error in JSON code';
  }
};

// Validate and process Key-Value data
const validateAndPreviewKeyValue = () => {
  try {
    const columns = JSON.parse(columnsDefinition.value);
    const rows = JSON.parse(rowsData.value);

    if (!Array.isArray(columns) || columns.length === 0) {
      isKeyValueValid.value = false;
      keyValueValidationError.value = 'Columns must be a non-empty array';
      return;
    }

    if (!Array.isArray(rows) || rows.length === 0) {
      isKeyValueValid.value = false;
      keyValueValidationError.value = 'Rows must be a non-empty array';
      return;
    }

    // Check for duplicate IDs in rows
    if (hasDuplicateIds(rows)) {
      isKeyValueValid.value = false;
      keyValueValidationError.value = 'Row data contains duplicate ID values';
      return;
    }

    // Check that all columns have name, label, and field properties
    const invalidColumns = columns.filter(col =>
      !col.name || !col.label || !col.field
    );

    if (invalidColumns.length > 0) {
      isKeyValueValid.value = false;
      keyValueValidationError.value = 'All columns must have name, label, and field properties';
      return;
    }

    // Check that all rows have all the fields defined in columns
    const requiredFields = columns.map(col => col.field);

    for (let i = 0; i < rows.length; i++) {
      const row = rows[i];
      const missingFields = requiredFields.filter(field => !(field in row));

      if (missingFields.length > 0) {
        isKeyValueValid.value = false;
        keyValueValidationError.value = `Row ${i+1} is missing required fields: ${missingFields.join(', ')}`;
        return;
      }
    }

    isKeyValueValid.value = true;
    keyValueValidationError.value = '';
    finalColumns.value = columns;
    finalData.value = rows;

  } catch (error) {
    isKeyValueValid.value = false;
    keyValueValidationError.value = 'Error in JSON code';
  }
};

// Validate all form data when disabled button is clicked
const validateAll = () => {
  // Check which validation needs to be run
  if (dataInputMethod.value === 'json' && !isJsonValid.value) {
    validateAndPreviewJson();
  } else if (dataInputMethod.value === 'keyValue' && !isKeyValueValid.value) {
    validateAndPreviewKeyValue();
  }

  // Show a notification about what's missing
  if (!reportTitle.value || !tableTitle.value) {
    $q.notify({
      color: 'negative',
      message: 'Please fill in all required fields (Report Title, Table Title)',
      icon: 'error'
    });
  } else if (!isJsonValid.value && !isKeyValueValid.value) {
    $q.notify({
      color: 'negative',
      message: 'Please validate your data before generating a report',
      icon: 'error'
    });
  } else if (includeNotes.value && (!notesTitle.value || !notesData.value)) {
    $q.notify({
      color: 'negative',
      message: 'Please complete the notes section or disable it',
      icon: 'error'
    });
  }
};

</script>

// Generate columns for the table preview
const generatedColumns = computed(() => {
  if (!reportConfig.tableData || !reportConfig.tableData.length) {
    return [];
  }

  // Get all unique keys from the first row
  const keys = Object.keys(reportConfig.tableData[0]);

  // Create column definitions for each key
  return keys.map(key => {
    return {
      name: key,
      label: formatLabel(key),
      field: key,
      align: reportConfig.rtl ? 'right' : 'left'
    };
  });
});

// Generate code for component
const generatedCode = computed(() => {
  // Create a simplified version of the config for display
  const config = { ...reportConfig };

  // Remove the tableData for brevity
  delete config.tableData;

  // Build the component code
  let code = '<pdf-export-component\n';

  // Add string props
  if (config.title) code += `  title="${config.title}"\n`;
  if (config.subtitle) code += `  subtitle="${config.subtitle}"\n`;
  if (config.tableTitle) code += `  tableTitle="${config.tableTitle}"\n`;
  if (config.summaryTitle) code += `  summaryTitle="${config.summaryTitle}"\n`;
  if (config.notesTitle) code += `  notesTitle="${config.notesTitle}"\n`;
  if (config.notes_data) code += `  notes_data="${config.notes_data}"\n`;
  if (config.buttonLabel) code += `  buttonLabel="${config.buttonLabel}"\n`;
  if (config.buttonColor !== 'primary') code += `  buttonColor="${config.buttonColor}"\n`;
  if (config.buttonIcon !== 'picture_as_pdf') code += `  buttonIcon="${config.buttonIcon}"\n`;
  if (config.buttonSize !== 'md') code += `  buttonSize="${config.buttonSize}"\n`;
  if (config.pdfFilename) code += `  pdfFilename="${config.pdfFilename}"\n`;

  // Add boolean props
  if (config.buttonFlat) code += '  buttonFlat\n';
  if (config.buttonOutline) code += '  buttonOutline\n';
  if (config.rtl) code += '  :rtl="true"\n';

  // Add object props
  code += '  :tableData="yourData"\n';

  if (config.summaryData && Object.keys(config.summaryData).length > 0) {
    code += `  :summaryData="${JSON.stringify(config.summaryData)}"\n`;
  }

  code += '/>';

  return code;
});

// Format column labels
function formatLabel(key) {
  // Convert camelCase or snake_case to Title Case
  return key
    .replace(/([A-Z])/g, ' $1') // Add space before capital letters
    .replace(/_/g, ' ')        // Replace underscores with spaces
    .replace(/^\w/, c => c.toUpperCase()) // Capitalize first letter
    .trim();                  // Remove extra spaces
}

// Add a new summary item
function addSummaryItem() {
  summaryKeys.value.push("");
  summaryValues.value.push("");
  updateSummaryData();
}

// Remove a summary item
function removeSummaryItem(index) {
  summaryKeys.value.splice(index, 1);
  summaryValues.value.splice(index, 1);
  updateSummaryData();
}

// Update the summaryData object
function updateSummaryData() {
  const summaryData = {};
  summaryKeys.value.forEach((key, index) => {
    if (key.trim()) {
      summaryData[key] = summaryValues.value[index];
    }
  });
  reportConfig.summaryData = summaryData;
}

// Copy the generated code to clipboard
function copyToClipboard() {
  navigator.clipboard.writeText(generatedCode.value).then(() => {
    $q.notify({
      message: 'Code copied to clipboard',
      color: 'positive',
      icon: 'content_copy',
      position: 'top'
    });
  }).catch(err => {
    $q.notify({
      message: 'Failed to copy code',
      color: 'negative',
      icon: 'error',
      position: 'top'
    });
    console.error('Failed to copy: ', err);
  });
}

// Watch for changes to summary data
watch(summaryKeys, () => {
  updateSummaryData();
}, { deep: true });

watch(summaryValues, () => {
  updateSummaryData();
}, { deep: true });
